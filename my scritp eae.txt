-- Safely download and run a button's script.
-- Requires an environment that supports HttpGet and loadstring.
local function runRemoteScript(url)
    if type(loadstring) ~= "function" then
        warn("Eukard's Hub: this environment does not support loadstring.")
        return
    end

    local requestOK, source = pcall(function()
        return game:HttpGet(url)
    end)

    if not requestOK or type(source) ~= "string" or source == "" then
        warn("Eukard's Hub: failed to download script: " .. tostring(url))
        warn(tostring(source))
        return
    end

    local compileOK, compiled = pcall(loadstring, source)
    if not compileOK or type(compiled) ~= "function" then
        warn("Eukard's Hub: script could not be compiled: " .. tostring(url))
        warn(tostring(compiled))
        return
    end

    local runOK, runError = pcall(compiled)
    if not runOK then
        warn("Eukard's Hub: script failed while running: " .. tostring(url))
        warn(tostring(runError))
    end
end
